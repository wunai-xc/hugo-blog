学生管理系统需要实现以下功能：

4. **数据持久化**
   - 保存数据到文件

```java
    FEMALE("女");

    private final String displayName;

    Gender(String displayName) {
        this.displayName = displayName;
    }

    public String getDisplayName() {
        return displayName;
    }

    public static Gender fromDisplayName(String displayName) {
        for (Gender g : Gender.values()) {
            if (g.displayName.equals(displayName)) {
                return g;
            }
        }
        throw new IllegalArgumentException("无效的性别: " + displayName);
    }

    @Override
    public String toString() {
        return displayName;
    }
```

```java
import java.io.*;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;

/**
 * 数据存储类 - 负责数据的持久化
 */
public class DataStorage {
    private static final String DATA_DIR = "data";
    private static final String STUDENT_FILE = DATA_DIR + File.separator + "students.dat";
    private static final String GRADE_FILE = DATA_DIR + File.separator + "grades.dat";
    private static final String CLASS_FILE = DATA_DIR + File.separator + "classes.dat";
    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.ofPattern("yyyy-MM-dd");

    /**
     * 保存所有数据
     */
    public static void saveData(Map<String, Student> students,
                                Map<String, List<Grade>> grades,
                                Map<String, ClassInfo> classes) throws IOException {
        createDataDirectory();

        saveStudents(students);
        saveGrades(grades);
        saveClasses(classes);
    }

    private static void createDataDirectory() {
        File dir = new File(DATA_DIR);
        if (!dir.exists()) {
            dir.mkdirs();
        }
    }

    /**
     * 保存学生数据
     */
    private static void saveStudents(Map<String, Student> students) throws IOException {
        try (ObjectOutputStream oos = new ObjectOutputStream(
                new FileOutputStream(STUDENT_FILE))) {
            oos.writeObject(students);
        }
    }

    /**
     * 保存成绩数据
     */
    private static void saveGrades(Map<String, List<Grade>> grades) throws IOException {
        try (ObjectOutputStream oos = new ObjectOutputStream(
                new FileOutputStream(GRADE_FILE))) {
            oos.writeObject(grades);
        }
    }

    /**
     * 保存班级数据
     */
    private static void saveClasses(Map<String, ClassInfo> classes) throws IOException {
        try (ObjectOutputStream oos = new ObjectOutputStream(
                new FileOutputStream(CLASS_FILE))) {
            oos.writeObject(classes);
        }
    }

    /**
     * 加载所有数据
     */
    @SuppressWarnings("unchecked")
    public static DataBundle loadData() {
        DataBundle bundle = new DataBundle();

        // 尝试加载学生数据
        File studentFile = new File(STUDENT_FILE);
        if (studentFile.exists()) {
            try (ObjectInputStream ois = new ObjectInputStream(
                    new FileInputStream(studentFile))) {
                bundle.students = (Map<String, Student>) ois.readObject();
            } catch (IOException | ClassNotFoundException e) {
                System.err.println("加载学生数据失败: " + e.getMessage());
                bundle.students = new HashMap<>();
            }
        } else {
            bundle.students = new HashMap<>();
        }

        // 尝试加载成绩数据
        File gradeFile = new File(GRADE_FILE);
        if (gradeFile.exists()) {
            try (ObjectInputStream ois = new ObjectInputStream(
                    new FileInputStream(gradeFile))) {
                bundle.grades = (Map<String, List<Grade>>) ois.readObject();
            } catch (IOException | ClassNotFoundException e) {
                System.err.println("加载成绩数据失败: " + e.getMessage());
                bundle.grades = new HashMap<>();
            }
        } else {
            bundle.grades = new HashMap<>();
        }

        // 尝试加载班级数据
        File classFile = new File(CLASS_FILE);
        if (classFile.exists()) {
            try (ObjectInputStream ois = new ObjectInputStream(
                    new FileInputStream(classFile))) {
                bundle.classes = (Map<String, ClassInfo>) ois.readObject();
            } catch (IOException | ClassNotFoundException e) {
                System.err.println("加载班级数据失败: " + e.getMessage());
                bundle.classes = new HashMap<>();
            }
        } else {
            bundle.classes = new HashMap<>();
        }

        return bundle;
    }

    /**
     * 数据包（用于返回加载的数据）
     */
    public static class DataBundle {
        public Map<String, Student> students;
        public Map<String, List<Grade>> grades;
        public Map<String, ClassInfo> classes;
    }
}
```

### 3.3 学生服务类：StudentService

```java
import java.time.LocalDate;
import java.util.*;
import java.util.stream.Collectors;

/**
 * 学生业务服务
 */
public class StudentService {
    private Map<String, Student> students;
    private Map<String, ClassInfo> classes;
    private Map<String, List<Grade>> grades;

    public StudentService(DataStorage.DataBundle bundle) {
        this.students = bundle.students;
        this.classes = bundle.classes;
        this.grades = bundle.grades;
    }

    /**
     * 添加学生
     */
    public void addStudent(String id, String name, String genderStr,
                           LocalDate birthDate, String className) throws ValidationException {
        // 校验学号
        if (id == null || id.trim().isEmpty()) {
            throw new ValidationException("id", "学号不能为空");
        }
        if (students.containsKey(id)) {
            throw new ValidationException("id", "学号已存在: " + id);
        }

        // 校验姓名
        if (name == null || name.trim().isEmpty()) {
            throw new ValidationException("name", "姓名不能为空");
        }

        // 校验性别
        Gender gender;
        try {
            gender = Gender.fromDisplayName(genderStr);
        } catch (IllegalArgumentException e) {
            throw new ValidationException("gender", "性别必须为'男'或'女'");
        }

        // 校验班级
        if (className == null || className.trim().isEmpty()) {
            throw new ValidationException("className", "班级不能为空");
        }

        // 创建学生
        Student student = new Student(id, name, gender, birthDate, className);
        students.put(id, student);

        // 更新班级信息
        ClassInfo classInfo = classes.computeIfAbsent(className, ClassInfo::new);
        classInfo.addStudent(id);

        // 初始化成绩列表
        grades.computeIfAbsent(id, k -> new ArrayList<>());
    }

    /**
     * 删除学生
     */
    public boolean deleteStudent(String id) throws ValidationException {
        if (id == null || id.trim().isEmpty()) {
            throw new ValidationException("id", "学号不能为空");
        }

        Student student = students.remove(id);
        if (student == null) {
            return false;
        }

        // 从班级中移除
        ClassInfo classInfo = classes.get(student.getClassName());
        if (classInfo != null) {
            classInfo.removeStudent(id);
        }

        // 删除成绩
        grades.remove(id);

        return true;
    }

    /**
     * 修改学生信息
     */
    public boolean updateStudent(String id, String name, String genderStr,
                                 LocalDate birthDate, String className) throws ValidationException {
        Student student = students.get(id);
        if (student == null) {
            return false;
        }

        // 处理班级变更
        String oldClassName = student.getClassName();
        if (!oldClassName.equals(className)) {
            // 从旧班级移除
            ClassInfo oldClass = classes.get(oldClassName);
            if (oldClass != null) {
                oldClass.removeStudent(id);
            }
            // 加入新班级
            ClassInfo newClass = classes.computeIfAbsent(className, ClassInfo::new);
            newClass.addStudent(id);
        }

        // 更新信息
        if (name != null && !name.trim().isEmpty()) {
            student.setName(name);
        }
        if (genderStr != null && !genderStr.trim().isEmpty()) {
            student.setGender(Gender.fromDisplayName(genderStr));
        }
        if (birthDate != null) {
            student.setBirthDate(birthDate);
        }
        student.setClassName(className);

        return true;
    }

    /**
     * 按学号查找学生
     */
    public Optional<Student> findById(String id) {
        return Optional.ofNullable(students.get(id));
    }

    /**
     * 按姓名查找学生
     */
    public List<Student> findByName(String name) {
        if (name == null || name.trim().isEmpty()) {
            return new ArrayList<>();
        }
        return students.values().stream()
                .filter(s -> s.getName().contains(name.trim()))
                .collect(Collectors.toList());
    }

    /**
     * 按班级查找学生
     */
    public List<Student> findByClass(String className) {
        if (className == null || className.trim().isEmpty()) {
            return new ArrayList<>();
        }
        return students.values().stream()
                .filter(s -> s.getClassName().equals(className))
                .collect(Collectors.toList());
    }

    /**
     * 获取所有学生
     */
    public List<Student> getAllStudents() {
        return new ArrayList<>(students.values());
    }

    /**
     * 获取所有班级列表
     */
    public List<ClassInfo> getAllClasses() {
        return new ArrayList<>(classes.values());
    }

    /**
     * 获取学生数量
     */
    public int getStudentCount() {
        return students.size();
    }

    /**
     * 获取各班级人数统计
     */
    public Map<String, Integer> getClassStatistics() {
        return classes.values().stream()
                .collect(Collectors.toMap(
                        ClassInfo::getName,
                        ClassInfo::getStudentCount,
                        (e1, e2) -> e1,
                        LinkedHashMap::new
                ));
    }

    /**
     * 获取当前数据
     */
    public Map<String, Student> getStudents() { return students; }
    public Map<String, ClassInfo> getClasses() { return classes; }
    public Map<String, List<Grade>> getGrades() { return grades; }
}
```

### 3.4 成绩服务类：GradeService

```java
import java.time.LocalDate;
import java.util.*;
import java.util.stream.Collectors;

/**
 * 成绩业务服务
 */
public class GradeService {
    private Map<String, Student> students;
    private Map<String, List<Grade>> grades;

    public GradeService(Map<String, Student> students, Map<String, List<Grade>> grades) {
        this.students = students;
        this.grades = grades;
    }

    /**
     * 添加成绩
     */
    public void addGrade(String studentId, String course, double score) throws ValidationException {
        // 校验学生是否存在
        if (!students.containsKey(studentId)) {
            throw new ValidationException("studentId", "学生不存在: " + studentId);
        }

        // 校验课程
        if (course == null || course.trim().isEmpty()) {
            throw new ValidationException("course", "课程名称不能为空");
        }

        // 校验分数
        if (score < 0 || score > 100) {
            throw new ValidationException("score", "分数必须在0-100之间");
        }

        // 检查是否已存在该课程成绩（同一课程只能有一个成绩）
        List<Grade> studentGrades = grades.computeIfAbsent(studentId, k -> new ArrayList<>());
        Optional<Grade> existing = studentGrades.stream()
                .filter(g -> g.getCourse().equals(course))
                .findFirst();

        if (existing.isPresent()) {
            throw new ValidationException("course", "该课程已存在成绩，请使用修改功能");
        }

        Grade grade = new Grade(studentId, course, score);
        studentGrades.add(grade);
    }

    /**
     * 修改成绩
     */
    public boolean updateGrade(String studentId, String course, double newScore) throws ValidationException {
        if (newScore < 0 || newScore > 100) {
            throw new ValidationException("score", "分数必须在0-100之间");
        }

        List<Grade> studentGrades = grades.get(studentId);
        if (studentGrades == null || studentGrades.isEmpty()) {
            return false;
        }

        Optional<Grade> optional = studentGrades.stream()
                .filter(g -> g.getCourse().equals(course))
                .findFirst();

        if (optional.isPresent()) {
            optional.get().setScore(newScore);
            return true;
        }

        return false;
    }

    /**
     * 删除成绩
     */
    public boolean deleteGrade(String studentId, String course) {
        List<Grade> studentGrades = grades.get(studentId);
        if (studentGrades == null || studentGrades.isEmpty()) {
            return false;
        }

        return studentGrades.removeIf(g -> g.getCourse().equals(course));
    }

    /**
     * 获取学生全部成绩
     */
    public List<Grade> getGradesByStudent(String studentId) {
        return grades.getOrDefault(studentId, new ArrayList<>());
    }

    /**
     * 获取某课程的所有成绩
     */
    public List<Grade> getGradesByCourse(String course) {
        List<Grade> result = new ArrayList<>();
        for (List<Grade> gradeList : grades.values()) {
            gradeList.stream()
                    .filter(g -> g.getCourse().equals(course))
                    .forEach(result::add);
        }
        return result;
    }

    /**
     * 计算学生平均分
     */
    public double getAverageScore(String studentId) {
        List<Grade> studentGrades = grades.get(studentId);
        if (studentGrades == null || studentGrades.isEmpty()) {
            return 0;
        }
        return studentGrades.stream()
                .mapToDouble(Grade::getScore)
                .average()
                .orElse(0);
    }

    /**
     * 获取课程平均分
     */
    public double getCourseAverage(String course) {
        List<Grade> courseGrades = getGradesByCourse(course);
        if (courseGrades.isEmpty()) {
            return 0;
        }
        return courseGrades.stream()
                .mapToDouble(Grade::getScore)
                .average()
                .orElse(0);
    }

    /**
     * 获取学生成绩排名（按指定课程）
     */
    public List<StudentGradeInfo> getRankingByCourse(String course) {
        List<Grade> courseGrades = getGradesByCourse(course);

        return courseGrades.stream()
                .map(g -> {
                    Student s = students.get(g.getStudentId());
                    return new StudentGradeInfo(
                            g.getStudentId(),
                            s != null ? s.getName() : "未知",
                            g.getScore()
                    );
                })
                .sorted((a, b) -> Double.compare(b.score, a.score))
                .collect(Collectors.toList());
    }

    /**
     * 获取所有课程的统计信息
     */
    public Map<String, CourseStatistics> getCourseStatistics() {
        // 收集所有课程
        Set<String> allCourses = new HashSet<>();
        for (List<Grade> gradeList : grades.values()) {
            for (Grade g : gradeList) {
                allCourses.add(g.getCourse());
            }
        }

        Map<String, CourseStatistics> stats = new LinkedHashMap<>();
        for (String course : allCourses) {
            List<Grade> courseGrades = getGradesByCourse(course);
            CourseStatistics cs = new CourseStatistics();
            cs.course = course;
            cs.count = courseGrades.size();
            cs.average = courseGrades.stream()
                    .mapToDouble(Grade::getScore)
                    .average()
                    .orElse(0);
            cs.max = courseGrades.stream()
                    .mapToDouble(Grade::getScore)
                    .max()
                    .orElse(0);
            cs.min = courseGrades.stream()
                    .mapToDouble(Grade::getScore)
                    .min()
                    .orElse(0);
            stats.put(course, cs);
        }

        return stats;
    }

    /**
     * 学生成绩信息（用于排名）
     */
    public static class StudentGradeInfo {
        public String studentId;
        public String name;
        public double score;

        public StudentGradeInfo(String studentId, String name, double score) {
            this.studentId = studentId;
            this.name = name;
            this.score = score;
        }

        @Override
        public String toString() {
            return String.format("%-6s | %-8s | %.2f", studentId, name, score);
        }
    }

    /**
     * 课程统计信息
     */
    public static class CourseStatistics {
        public String course;
        public int count;
        public double average;
        public double max;
        public double min;

        @Override
        public String toString() {
            return String.format("课程: %-10s | 人数: %3d | 平均: %6.2f | 最高: %6.2f | 最低: %6.2f",
                    course, count, average, max, min);
        }
    }
```

```java
            DataStorage.DataBundle bundle = DataStorage.loadData();
            // 更新服务中的数据
            studentService.getStudents().clear();
            studentService.getStudents().putAll(bundle.students);
            studentService.getClasses().clear();
            studentService.getClasses().putAll(bundle.classes);
            studentService.getGrades().clear();
            studentService.getGrades().putAll(bundle.grades);
            System.out.println("✅ 数据加载成功！");
        } catch (Exception e) {
            System.out.println("⚠️ 数据加载失败: " + e.getMessage());
        }
    }
}
```
